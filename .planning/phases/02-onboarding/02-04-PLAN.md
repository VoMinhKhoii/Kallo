**3. Create `components/onboarding/onboarding-prompt.tsx`** — `'use client'`:
This is the SSR-safe wrapper that mounts both card and nudge on the home page.
- Props: `{ onboardingStep: number; accountCreatedAt: string | Date }`
- **Card dismiss state with 7-day expiry (SSR-safe):**
  ```ts
  const CARD_DISMISS_KEY = 'onboarding_card_dismissed_at';
  const CARD_DISMISS_EXPIRY_MS = 7 * 24 * 60 * 60 * 1000; // 7 days

  const [cardDismissed, setCardDismissed] = useState(false);

  useEffect(() => {
    const dismissedAt = localStorage.getItem(CARD_DISMISS_KEY);
    if (dismissedAt) {
      const elapsed = Date.now() - parseInt(dismissedAt, 10);
      setCardDismissed(elapsed < CARD_DISMISS_EXPIRY_MS);
    }
  }, []);

  function handleDismissCard() {
    localStorage.setItem(CARD_DISMISS_KEY, Date.now().toString());
    setCardDismissed(true);
  }
  ```
- Render:
  ```tsx
  return (
    <>
      {!cardDismissed && (
        <OnboardingCard
          onboardingStep={onboardingStep}
          onDismiss={handleDismissCard}
        />
      )}
      <NudgeDialog
        onboardingStep={onboardingStep}
        accountCreatedAt={accountCreatedAt}
      />
    </>
  );
  ```
